    =>  Flat an Multi Dimension Array:-
        const val = [1,2,[3,[5,4]]]

        # 1st way
        console.log(val.flat(Infinity));

        # 2nd way
        function flattenArray(arr) {
            let result = [];
            arr.forEach(element => {
                if (Array.isArray(element)) {
                    result = [ ...result ,...flattenArray(element)];
                } else {
                    result.push(element);
                }
            });
            return result;
        }

=>  For Loop:- 

    =>  For In:- loops through the properties of an Object
        Ex:- 
            const person = {fname:"John", lname:"Doe", age:25};
            for (let x in person) {
                console.log(person[x]);  # John , Doe
            }
    
    =>  For Of:- loops through the values of an iterable object  ( Array , String )
        Ex:- 
            const cars = ["BMW", "Volvo", "Mini"];
            for (let x of cars) {
                console.log(x);
            }
