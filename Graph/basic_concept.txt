    // With Recursion
        class Solution {
        private: 
            void dfs(int node, vector<int> adj[], int vis[], vector<int> &ls) {
                vis[node] = 1; 
                ls.push_back(node); 
                // traverse all its neighbours
                for(auto it : adj[node]) {
                    // if the neighbour is not visited
                    if(!vis[it]) {
                        dfs(it, adj, vis, ls); 
                    }
                }
            }
        public:
            // Function to return a list containing the DFS traversal of the graph.
            vector<int> dfsOfGraph(int V, vector<int> adj[]) {
                int vis[V] = {0}; 
                int start = 0;
                // create a list to store dfs
                vector<int> ls; 
                // call dfs for starting node
                dfs(start, adj, vis, ls); 
                return ls; 
            }
        };

    // Iterative Apporach:-
        void Graph::DFS(int s , vector<vector<int>>&adj)
        {
            vector<bool> visited(V, false);
        
            // Create a stack for DFS
            stack<int> stack;
            stack.push(s);
            while (!stack.empty())
            {
                // Pop a vertex from stack and print it
                int s = stack.top();
                stack.pop();

                if (!visited[s])
                {
                    visited[s] = true;
                }
        
                // Get all adjacent vertices of the popped vertex s
                // If a adjacent has not been visited, then push it
                // to the stack.
                for (auto i = adj[s].begin(); i != adj[s].end(); ++i)
                    if (!visited[*i])
                        stack.push(*i);
            }
        }


------------------------------- Breadth First Search (BFS): Level Order Traversal ------------------------------------------->


    vector<int> bfsOfGraph(int V, vector<int> adj[]) {
        int vis[V] = {0}; 
        vis[0] = 1; 
        queue<int> q;
        // push the initial starting node 
        q.push(0); 
        vector<int> bfs; 
        // iterate till the queue is empty 
        while(!q.empty()) {
           // get the topmost element in the queue 
            int node = q.front(); 
            q.pop(); 
            bfs.push_back(node); 
            // traverse for all its neighbours 
            for(auto it : adj[node]) {
                // if the neighbour has previously not been visited, 
                // store in Q and mark as visited 
                if(!vis[it]) {
                    vis[it] = 1; 
                    q.push(it); 
                }
            }
        }
        return bfs; 
    }

Question 4:- G-35 : Print Shortest Path - Dijkstra’s Algorithm

Problem Statement: 
    You are given a weighted undirected graph having n+1 vertices numbered from 0 to n and m edges describing there are edges 
    between a to b with some weight, find the shortest path between the vertex 1 and the vertex n, and if the path does not exist
    then return a list consisting of only -1.

LINK:- https://www.geeksforgeeks.org/problems/implementing-dijkstra-set-1-adjacency-matrix/1

NOTE:-  Dijkstra’s Algorithm:-
       There are 3 main thing we need for shortest path ( Dijkstra’s ). Min heap ( priority queue ), parent vector ( to keep 
        from where we are coming parent node ) and distance vector ( store min distance to reach that particular node ).


Approach/Intitution:-

1. First we declare dist[source] with 0 value . Then we will push the node with 0 dist into min heap. Then start travering its 
    adjacency nodes.
2. while iterating we check if parent node dist + adjNode distance > dist[adjNode]. That mean there is a shorter path present. So, 
    we will update the dist vector. 
3. Atlast we check if destination node == 1e9 . It means there is no node reachable to destination node.

// CODE LOGIC:- 
    vector<int> shortestPath(int n, int m, vector<vector<int>> &edges)
    {
        // Create an adjacency list of pairs of the form node1 -> {node2, edge weight}
        // where the edge weight is the weight of the edge from node1 to node2.
        vector<pair<int, int>> adj[n + 1];
        for (auto it : edges)
        {
            adj[it[0]].push_back({it[1], it[2]});
            adj[it[1]].push_back({it[0], it[2]});
        }
        // Create a priority queue for storing the nodes along with distances in the form of a pair { dist, node }.
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int,int>>> pq;

        // Create a dist array for storing the updated distances and a parent array
        //for storing the nodes from where the current nodes represented by indices of
        // the parent array came from.
        vector<int> dist(n + 1, 1e9), parent(n + 1);
        for (int i = 1; i <= n; i++)
            parent[i] = i;

        dist[1] = 0;

        // Push the source node to the queue.
        pq.push({0, 1});
        while (!pq.empty())
        {
            // Topmost element of the priority queue is with minimum distance value.
            auto it = pq.top();
            pq.pop();
            int node = it.second;
            int dis = it.first;

            // Iterate through the adjacent nodes of the current popped node.
            for (auto it : adj[node])
            {
                int adjNode = it.first;
                int edW = it.second;

                // Check if the previously stored distance value is 
                // greater than the current computed value or not, 
                // if yes then update the distance value.
                if (dis + edW < dist[adjNode])
                {
                    dist[adjNode] = dis + edW;
                    pq.push({dis + edW, adjNode});

                    // Update the parent of the adjNode to the recent 
                    // node where it came from.
                    parent[adjNode] = node;
                }
            }
        }

        // If distance to a node could not be found, return an array containing -1.
        if (dist[n] == 1e9)
            return {-1};

        // Store the final path in the ‘path’ array.
        vector<int> path;
        int node = n;

        // Iterate backwards from destination to source through the parent array.
        while (parent[node] != node)
        {
            path.push_back(node);
            node = parent[node];
        }
        path.push_back(1);

        // Since the path stored is in a reverse order, we reverse the array
        // to get the final answer and then return the array.
        reverse(path.begin(), path.end());
        return path;
    }

Question 5: Bellman Ford Algorithm: G-41
Problem Statement: Given a weighted, directed and connected graph of V vertices and E edges, Find the shortest distance of 
                all the vertices from the source vertex S.

LINK:- https://www.geeksforgeeks.org/problems/distance-from-the-source-bellman-ford-algorithm/

NOTE:- 
    1) The bellman-Ford algorithm helps to find the shortest distance from the source node to all other nodes. 
      But, we have already learned Dijkstra's algorithm (Dijkstra's algorithm article link) to fulfill the same purpose.
      Now, the question is how this algorithm is different from Dijkstra's algorithm.

    2) While learning Dijkstra's algorithm, we came across the following two situations, where Dijkstra's algorithm failed:
       1. If the graph contains negative edges.
       2. If the graph has a negative cycle (In this case Dijkstra's algorithm fails to minimize the distance,
            keeps on running, and goes into an infinite loop. As a result it gives TLE error).

    
    3) Negative Cycle: A cycle is called a negative cycle if the sum of all its weights becomes negative. 


Intitution/Approach:- 
    1. In this algorithm, the edges can be given in any order. The intuition is to relax all the edges for N-1( N = no. of nodes)
        times sequentially. After N-1 iterations, we should have minimized the distance to every node.
    2. Again we iteration Nth times we check if anything is updating . It means there is a negative cycle.
    

    
Q1:- why N-1 iteration ?
-> If we try to iterate Nth iteration the path weight will be decreased in every iteration and in wrost case we need to iteration N-1
    iteration.

    Note: Points to remember since, in a graph of N nodes we will take at most N-1 edges to reach from the first to the last node,
        we need exact N-1 iterations. It is impossible to draw a graph that takes more than N-1 edges to reach any node. 

Q2:- How to detect a negative cycle in the graph ?
-> In order to check the existence of a negative cycle, we will relax the edges one more time after the completion of N-1 iterations.
    And if in that Nth iteration, it is found that further relaxation of any edge is possible, we can conclude that the graph has 
    a negative cycle. Thus, the Bellman-Ford algorithm detects negative cycles.


// CODE LOGIC:-
    vector<int> bellman_ford(int V, vector<vector<int>>& edges, int S) {
		vector<int> dist(V, 1e8);
		dist[S] = 0;
		for (int i = 0; i < V - 1; i++) {
			for (auto it : edges) {
				int u = it[0];
				int v = it[1];
				int wt = it[2];
				if (dist[u] != 1e8 && dist[u] + wt < dist[v]) {
					dist[v] = dist[u] + wt;
				}
			}
		}
		// Nth relaxation to check negative cycle
		for (auto it : edges) {
			int u = it[0];
			int v = it[1];
			int wt = it[2];
			if (dist[u] != 1e8 && dist[u] + wt < dist[v]) {
				return { -1};
			}
		}


		return dist;
	}

Question 6: Floyd Warshall Algorithm: G-42
Problem Statement: The problem is to find the shortest distances between every pair of vertices in a given 
        edge-weighted directed graph. The graph is represented as an adjacency matrix of size n*n. 
        Matrix[i][j] denotes the weight of the edge from i to j. If Matrix[i][j]=-1, it means there is no
        edge from i to j.

LINK:- https://www.geeksforgeeks.org/problems/implementing-floyd-warshall2042/1

NOTE:- 1. Floyd Warshall algorithm is a multi-source shortest path algorithm and it helps to detect negative cycles as well.
       2. It works on both directed and undirected graph ( need to just convert it into directed graph).

Question:- What will happen if we will apply Dijkstra’s algorithm for this purpose?
        1. If the graph has a negative cycle: We cannot apply Dijkstra’s algorithm to the graph which contains
            a negative cycle. It will give TLE error in that case.
        2. If the graph does not contain a negative cycle: In this case, we will apply Dijkstra’s algorithm 
            for every possible node to make it work like a multi-source shortest path algorithm like Floyd Warshall.
            The time complexity of Floyd Warshall is O(V3)(Which we will discuss later in this article) whereas if we
            apply Dijkstra’s algorithm for the same purpose the time complexity reduces to O(V*(E*logV)) 
            (where v = no. of vertices). 

// Intitution / Approach:-
1. We are going to count all shortest path from 0 to n size. 
2. Path (a,b) is like min(path(a,b) , (path(a,c) + path (c,b)) ). c will be the size of matrix 0 to n.
3. To detect negative cycle , we check if there is any negative edges after computing shortest path.


// CODE LOGIC:-

    void shortest_distance(vector<vector<int>>&matrix) {
		int n = matrix.size();
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (matrix[i][j] == -1) {
					matrix[i][j] = 1e9;
				}
				if (i == j) matrix[i][j] = 0;
			}
		}

		for (int k = 0; k < n; k++) {
			for (int i = 0; i < n; i++) {
				for (int j = 0; j < n; j++) {
                    // To compute path from 0 to n
					matrix[i][j] = min(matrix[i][j], matrix[i][k] + matrix[k][j]);
				}
			}
		}




		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (matrix[i][j] == 1e9) {
					matrix[i][j] = -1;
				}
			}
		}
	}

// Approach / Intitution:-
1. We need priority_queue ( min heap ) to keep track track of min weighted edge.
2. Visited Array to keep track of visited node and a mst vector to store the edges while calculating MST edges.
3. First we insert any node with with {0,0}. While we keep checking if it is already visited then continue. 
   Else mark it as visited , add its weight into ans and keep iterating its adj, push it into queue.

// CODE LOGIC:- 

    int spanningTree(int V, vector<vector<int>> adj[])
    {
        // code here
        priority_queue<pair<int, int>, vector<pair<int,int>>, greater<pair<int, int>>>pq;

        vector<int>vis(V,0);
        // vector<pair<int,int>>mst;
        int sum =0;
        pq.push({0,0});
        while(!pq.empty()){
            auto cur = pq.top();
            int wt = cur.first;
            int node= cur.second;
            // int parent = cur.second.second;
            pq.pop();
            if(vis[node] == 1) continue;
            vis[node]=1;
            sum+=wt;
            // mst.push_back({node,parent});
            for(auto it:adj[node]){
                if(!vis[it[0]]){
                    pq.push({it[1],it[0]});
                }
            }
        }
        return sum;
    }

Question 8:- Disjoint Set | Union by Rank | Union by Size | Path Compression: G-46
problem statement: Given two components of an undirected graph
                The question is whether node 1 and node 5 are in the same component or not.

LINK: https://www.geeksforgeeks.org/problems/disjoint-set-union-find/1

// Approach / Intitution:-
1. By using DisjointSet , we are going to define all parent value to it self value and size will be 1 for all nodes.
2. In findUPar, takes a single node as an argument and finds the ultimate parent for each node.
3. In unionByRank, we check every node whose rank is greater than other we will merge smaller one into larger and if 
    both are equal first we merge it, then increase its rank.
4. In unionBySize , this is better than unionByRank. We will connect the ultimate parent with a smaller size to
    the other ultimate parent with a larger size. But if the size of the two is equal, we can connect any parent
    to the other parent. while connecting we will the size of merging parent with smaller parent size.

Question ->Difference between  unionByRank and unionBySize ?
-> 1. It seems much more intuitive than union by rank as the rank gets distorted after path compression.
   2. This is as same as the Union by rank method except this method uses the size to compare the components while
        connecting. That is why we need a ‘size’ array of size N(no. of nodes) instead of a rank array. The size 
        array will be storing the size for each particular node i.e. size[i] will be the size of the component 
        starting from node i.

// CODE LOGIC:-
    class DisjointSet {
    vector<int> rank, parent,size;
    public:
        DisjointSet(int n) {
            rank.resize(n + 1, 0);
            parent.resize(n + 1);
            size.resize(n + 1);
            for (int i = 0; i <= n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }

        int findUPar(int node) {
            if (node == parent[node])
                return node;
            return parent[node] = findUPar(parent[node]);
        }

        void unionByRank(int u, int v) {
            int ulp_u = findUPar(u);
            int ulp_v = findUPar(v);
            if (ulp_u == ulp_v) return;
            if (rank[ulp_u] < rank[ulp_v]) {
                parent[ulp_u] = ulp_v;
            }
            else if (rank[ulp_v] < rank[ulp_u]) {
                parent[ulp_v] = ulp_u;
            }
            else {
                parent[ulp_v] = ulp_u;
                rank[ulp_u]++;
            }
        }

        void unionBySize(int u, int v) {
            int ulp_u = findUPar(u);
            int ulp_v = findUPar(v);
            if (ulp_u == ulp_v) return;
            if (size[ulp_u] < size[ulp_v]) {
                parent[ulp_u] = ulp_v;
                size[ulp_v] += size[ulp_u];
            }
            else {
                parent[ulp_v] = ulp_u;
                size[ulp_u] += size[ulp_v];
            }
        }
    };

    int main() {
        DisjointSet ds(7);
        ds.unionBySize(1, 2);
        ds.unionBySize(2, 3);
        ds.unionBySize(4, 5);
        ds.unionBySize(6, 7);
        ds.unionBySize(5, 6);
        // if 3 and 7 same or not
        if (ds.findUPar(3) == ds.findUPar(7)) {
            cout << "Same\n";
        }
        else cout << "Not same\n";

        ds.unionBySize(3, 7);

        if (ds.findUPar(3) == ds.findUPar(7)) {
            cout << "Same\n";
        }
        else cout << "Not same\n";
        return 0;
    }

Question 9:- Kruskal's Algorithm - Minimum Spanning Tree : G-47
Problem Statement: Given a weighted, undirected, and connected graph of V vertices and E edges. The task is to find 
    the sum of weights of the edges of the Minimum Spanning Tree.

LINK:- https://www.geeksforgeeks.org/problems/minimum-spanning-tree/1

// Approach / Intitution:-
1. We will be implementing Kruskal’s algorithm using the Disjoint Set data structure that we have previously learned.
2. We are going to create one vector with pair< weight ,pair< parent node ,adjNode>> and push all its edges.
3. We are going to sort based on weight . Then while iterating, we are going to check if parent node or indUPar(adjNode)
    not equal . Then add it into mst sum and call unionBySize for both nodes.

// CODE LOGIC:- 
    class DisjointSet {
        vector<int> rank, parent, size;
    public:
        DisjointSet(int n) {
            .....
        }
    };
    int spanningTree(int V, vector<vector<int>> adj[])
    {
        // 1 - 2 wt = 5
        // 1 - > (2, 5)
        // 2 -> (1, 5)

        vector<pair<int, pair<int, int>>> edges;
        for (int i = 0; i < V; i++) {
            for (auto it : adj[i]) {
                int adjNode = it[0];
                int wt = it[1];
                int node = i;

                edges.push_back({wt, {node, adjNode}});
            }
        }
        DisjointSet ds(V);
        sort(edges.begin(), edges.end());
        int mstWt = 0;
        for (auto it : edges) {
            int wt = it.first;
            int u = it.second.first;
            int v = it.second.second;

            if (ds.findUPar(u) != ds.findUPar(v)) {
                mstWt += wt;
                ds.unionBySize(u, v);
            }
        }

        return mstWt;
    }
