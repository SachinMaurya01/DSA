Question 1:- Matrix Chain Multiplication | (DP-48)
Problem Statement: Given a sequence of matrices, find the most efficient way to multiply these matrices together. 
        The efficient way is the one that involves the least number of multiplications. The dimensions of the
        matrices are given in an array arr[] of size n (such that n = number of matrices + 1) where the ith matrix
        has the dimensions (arr[i-1] x arr[i]).

LINK:- https://www.geeksforgeeks.org/problems/matrix-chain-multiplication0303/1

// Intitution / Approach :-

    1. We have an array of numbers and we are going to divide into 2 paritions like this (ABC) -> (A)(BC)
    2. In every , we have start and end point. We are going to iterate and in every k point we divide the partition into two parts
        and call recusive funtion to calculate sub array part.
    3. Example: arr = [2, 1, 3, 4];
        (i,j)-> (1,3)
        Logic:-  arr[i-1]*arr[k]*arr[j] + find(arr,i,k) + find(arr,k+1,j); 
        // arr[k]*arr[j] means [1,3,4] -> (1*3)*(3*4) if we calculate the matrix . We will get ( 1*4 ) -> Check Notes Logic

    // Note:- 
        1. Two matrices A1 and A2 of dimensions [p x q] and [r x s] can only be multiplied if and only if q=r.
        2. The total number of multiplications required to multiply A1 and A2 are: p*q*s

// Code Logic:-  
    int find(vector<int> &arr , int i, int j,vector<vector<int>>&dp){
        if(i==j) return 0;  // Base case if there is only element in array
        int steps=INT_MAX;
        if(dp[i][j]!= -1) return dp[i][j];
        for(int k=i;k<j;k++){
            int mini = arr[i-1]*arr[k]*arr[j] + find(arr,i,k,dp) + find(arr,k+1,j,dp);
            steps= min(steps,mini);
        }
        return dp[i][j]=steps;
    }
    int matrixMultiplication(vector<int> &arr) {
        int n=arr.size();
        vector<vector<int>>dp(n,vector<int>(n,-1));
        return find(arr,1,n-1,dp);
    }

Question 2:- Minimum Path Sum In a Grid (DP 10)
Problem Description: 
    We are given an “N*M” matrix of integers. We need to find a path from the top-left corner to the bottom-right
    corner of the matrix, such that there is a minimum cost past that we select.At every cell, we can move in only
    two directions: right and bottom. The cost of a path is given as the sum of values of cells of the given matrix.

LINK:- https://leetcode.com/problems/minimum-path-sum/description/

// Intitution / Approach :-
    1. We are going to use top-down approach ( start from last and climb till top element ).
    2. From bottom element call recursion for row - 1 (top) and col - 1 (left) and find min between 
        left and top calls including current element.
    3. Our Base case will be if its reaches top then return top element and if it goes out of bound
        then return INT_MAX . So, that we will not consider this solution.

// Code Logic:- 
    int minSumPathUtil(int i, int j, vector<vector<int>> &matrix, vector<vector<int>> &dp) {
        // Base cases
        if (i == 0 && j == 0)
            return matrix[0][0]; // If we are at the top-left corner, the minimum path sum is the value at (0, 0)
        if (i < 0 || j < 0)
            return INT_MAX; // If we go out of bounds, return a large value to avoid considering this path
        if (dp[i][j] != -1)
            return dp[i][j]; // If the result is already computed, return it

        // Calculate the minimum sum path by considering moving up and moving left
        int up = matrix[i][j] + minSumPathUtil(i - 1, j, matrix, dp);
        int left = matrix[i][j] + minSumPathUtil(i, j - 1, matrix, dp);

        // Store the result in the DP table and return it
        return dp[i][j] = min(up, left);
    }

    int minSumPath(int n, int m, vector<vector<int>> &matrix) {
        vector<vector<int>> dp(n, vector<int>(m, -1)); // DP table to memoize results
        return minSumPathUtil(n - 1, m - 1, matrix, dp); // Start from the bottom-right corner
    }

Question 3:- Subset sum equal to target (DP- 14)
Problem Description: 
    Given an integer array nums, return true if you can partition the array into two subsets such that the sum of the
    elements in both subsets is equal or false otherwise.

LINK:- https://leetcode.com/problems/partition-equal-subset-sum/description/

// Intitution / Approach :-
1. To find equal subset sum of array, we need to find s1 only (totalsum = s1 + s2) . If we find s1 , then logically
    remaining value sum is equal to s2. 
2. To find s1, we choose same knapsack solution. We have two option to choose that number or not to achieve target value.
3. In return statement , if either of taken or notTaken is true. It means there is subset sum which is equal to s1/s2.

// Code Logic:- 

    // -------------------------------- Memorization ---------------------->
    bool subsetSumUtil(int ind, int target, vector<int>& arr, vector<vector<int>>& dp) {
        // If the target sum is 0, we have found a subset
        if (target == 0)
            return true;

        // If we have reached the first element in 'arr'
        if (ind == 0)
            return arr[0] == target;

        // If the result for this subproblem has already been computed, return it
        if (dp[ind][target] != -1)
            return dp[ind][target];

        // Try not taking the current element into the subset
        bool notTaken = subsetSumUtil(ind - 1, target, arr, dp);

        // Try taking the current element into the subset if it doesn't exceed the target
        bool taken = false;
        if (arr[ind] <= target)
            taken = subsetSumUtil(ind - 1, target - arr[ind], arr, dp);

        // Store the result in the dp array to avoid recomputation
        return dp[ind][target] = notTaken || taken;
    }

    // -------------------------------- Memorization ---------------------->
    bool subsetSumToK(int n, int k, vector<int> &arr) {
        // Initialize a 2D DP array with dimensions (n x k+1) to store subproblem results
        vector<vector<bool>> dp(n, vector<bool>(k + 1, false));

        // Base case: If the target sum is 0, we can always achieve it by taking no elements
        for (int i = 0; i < n; i++) {
            dp[i][0] = true;
        }

        // Base case: If the first element of 'arr' is less than or equal to 'k', set dp[0][arr[0]] to true
        if (arr[0] <= k) {
            dp[0][arr[0]] = true;
        }

        // Fill the DP array iteratively
        for (int ind = 1; ind < n; ind++) {
            for (int target = 1; target <= k; target++) {
                // If we don't take the current element, the result is the same as the previous row
                bool notTaken = dp[ind - 1][target];

                // If we take the current element, subtract its value from the target and check the previous row
                bool taken = false;
                if (arr[ind] <= target) {
                    taken = dp[ind - 1][target - arr[ind]];
                }

                // Store the result in the DP array for the current subproblem
                dp[ind][target] = notTaken || taken;
            }
        }

        // The final result is stored in dp[n-1][k]
        return dp[n - 1][k];
    }

    bool canPartition(vector<int>& nums) {
        int n=nums.size();
        int sum = accumulate(nums.begin(),nums.end(),0);
        if(sum % 2 != 0) return false;
        int target = sum/2;
        vector<vector<int>>dp(n,vector<int>(target+1,-1));
        return helper(nums,n-1,target,dp);
    }

Question 4:-  Minimum Cost to Cut a Stick
Problem Description: 
    Given an integer array cuts where cuts[i] denotes a position you should perform a cut at.You should perform
    the cuts in order, you can change the order of the cuts as you wish. The cost of one cut is the length of the
    stick to be cut, the total cost is the sum of costs of all cuts. When you cut a stick, it will be split into 
    two smaller sticks

LINK:- https://leetcode.com/problems/minimum-cost-to-cut-a-stick/description/

// Intitution / Approach :-
    1. We are going to sort the cuts array. So, that from whereever we start cutting if left and left portion will
        symmetrical.
    2. We are going to iterate from 0 to n( length of stick ). In each cut we are going to calculate its
        left cost ( 0 to cuts[i]) and right cost ( cuts[i] to n).
    3. For each cut , we have end_stick - start_stick ( before cutting the stick length ) + left_cost + right_cost and 
        we are going to store min cost.

// Code Logic:- 

    // Memorization
        int dp[102][102];
        int solve(vector<int>&cuts, int left,int right){
            if(right - left <= 1) return 0;
            if(dp[left][right] != -1) return dp[left][right];

            int cost = 1e9;
            for(int i=left+1;i<=right-1;i++){
                int left_cost = solve(cuts,left,i);
                int right_cost = solve(cuts,i,right);
                int cur_cost = (cuts[right] - cuts[left]) + left_cost + right_cost;
                cost = min(cost,cur_cost);
            }
            return dp[left][right] = cost;
        }
        int minCost(int n, vector<int>& cuts) {
            memset(dp,-1,sizeof(dp));
            cuts.push_back(0);
            cuts.push_back(n);
            sort(cuts.begin(),cuts.end());
            return solve(cuts,0,cuts.size()-1);
        }

Question 5:-  Longest Palindromic Subsequence
Problem Description: 
    Given a string s, find the longest palindromic subsequence's length in s.

LINK:- https://leetcode.com/problems/longest-palindromic-subsequence/description/


// Intitution / Approach :-

    // Memorization :
        1. We use pointer start = 0 and end = s.size(). Then, we will keep check if first and last element
            matched . If yes , then add 2 and check for next element by decrementing and increment both value.
        2. Else check with next element ( solve(s, i+1 , j) , solve(s , i, j-1) );

    // Tabulation :
        1. We will initialise both starting dp value as 0 and create a new reverse string of s.
        2. Then we nested iterate for( 0 to n ) and check 
            if s[i-1]==r[j-1] :
                dp[i][j]=1+dp[i-1][j-1];
            else pick max of prev top and curr previous value:
                dp[i][j] = max(dp[i-1][j] , dp[i][j-1]);
        3. At last , we will return dp[n][n];

    

// Code Logic:- 
    // Memorization ( top-down )
    int solve(string &s , int i , int j){
        if(i>j) return 0 ;
        if(i==j) return 1 ;
        if(t[i][j] != -1) return t[i][j];
        if(s[i] == s[j]){
            return t[i][j] = 2 + solve(s , i+1 , j-1);
        }
        else{
            return t[i][j] = max(solve(s, i+1 , j) , solve(s , i, j-1));
        }
    }

    int longestPalindromeSubseq(string s) {
        int n=s.size();
        // -------------> Tabulation
        // vector<vector<int>>dp(n+1,vector<int>(n+1,-1));
        // return helper(s,i,s.size()-1,dp);

        string r = s;
        reverse(s.begin(),s.end());

        // ------------> Memory Optimisation
        vector<int>prev(n+1,-1);
        for(int i=0;i<n+1;i++){
            // dp[0][i]=0;
            // dp[i][0]=0;
            prev[i]=0;
        }

        for(int i=1;i<n+1;i++){
            vector<int>cur(n+1);
            for( int j=1;j<n+1;j++){
                if(s[i-1]==r[j-1]){
                    // dp[i][j]=1+dp[i-1][j-1];
                    cur[j]=1+prev[j-1];
                }else{
                    // dp[i][j]=max(dp[i-1][j] , dp[i][j-1]);
                    cur[j]= max(prev[j],cur[j-1]);
                }
            }
            prev=cur;
        }
        return prev[n];
    }

Question 5:- Max Sum Increasing Subsequence

Problem Description: 
    Given an array of positive integers arr. Find the maximum sum subsequence of the given array such 
    that the integers in the subsequence are sorted in strictly increasing order i.e. a strictly increasing subsequence. 

LINK:- https://www.geeksforgeeks.org/problems/maximum-sum-increasing-subsequence4749/1


// Intitution / Approach :-

    1. There is only one value is changing in array. So, we will create 1D array .
    2. We will loop through the from i to n and j to i . To check if there any increasing Subsequence exist.
    3. If we found, we will simply update the dp[i];

// Code Logic:- 

    // Memorization 
        int helper(vector<int>& arr ,vector<vector<int>>&dp, int indx,int prev ){
            if(indx == arr.size()) return 0;
            if(dp[indx][prev+1] != -1){
                return dp[indx][prev+1];
            }
            
            int notTaken = helper(arr,dp,indx+1,prev);
            int taken = 0;
            if(prev == -1 || arr[prev] < arr[indx]){
                taken = arr[indx] + helper(arr,dp,indx+1,indx);
            }
            int ans= max(taken , notTaken);
            return dp[indx][prev+1] = ans;
        }
        int maxSumIS(vector<int>& arr) {
            // Your code goes here
            int n=arr.size();
            vector<vector<int>>dp(n+1, vector<int>(n+1,-1));
            return helper(arr,dp,0,-1);
        }

    // Bottom Up:-  
        int maxSumIS(vector<int>& arr) {
            // Your code goes here
            int n=arr.size();
            int ans = 0;
        
            vector<int> dp(arr.size());
            
            // Compute maximum sum for each 
            // index and compare it with ans.
            for (int i=0; i<arr.size(); i++) {
                dp[i] = arr[i];
                
                for (int j=0; j<i; j++) {
                    if (arr[j]<arr[i]) {
                        dp[i] = max(dp[i], arr[i]+dp[j]);
                    }
                }
                
                ans = max(ans, dp[i]);
            }
            
            return ans;
            
        }
