
1. Right/Left view of binary tree
    Problem Statement: The Right View of a Binary Tree is a list of nodes that can be seen when the tree
    is viewed from the right side. The Left View of a Binary Tree is a list of nodes that can be seen when
    the tree is viewed from the left side.

Link :- https://www.geeksforgeeks.org/problems/left-view-of-binary-tree/1

Intitution / Approach :-

1. we can solve it by using bfs , pick Oth element for left view and last element for right view.
2. For Optimal, traversal right side for right view and keep track of lvl at same time. 
   Store the element if level is equal to the level.

=> Brute Force

    queue < Node *> q;
    q.push(root);

    while (!q.empty()) {
        int size = q.size();
        vector < int > level;
        for (int i = 0; i < size; i++) {
            Node * top = q.front();
            level.push_back(top -> data); // Pick first element for left view and last for right view
            q.pop();
            if (top -> left != NULL) q.push(top -> left);
            if (top -> right != NULL) q.push(top -> right);
        }
    }

 => Optimal Approach

    void recursionRight(Node* root, int level, vector<int> &res){
        if(root == NULL) return;
 	    // Check if the size of the result vector is equal to the current level
        if(res.size() == level){
            res.push_back(root->data); 
        }
        recursionRight(root->right, level + 1, res);
        recursionRight(root->left, level + 1, res);
    }



2. Bottom view of a Binary Tree
   Problem Statement: Given a Binary Tree, return its Bottom View. The Bottom View of
   a Binary Tree is the set of nodes visible when we see the tree from the bottom.

   LINK:- https://www.geeksforgeeks.org/problems/bottom-view-of-binary-tree/1

   Intitution / Approach :-
   1. We can use map to store the last value of particular line. 

    vector<int> bottomView(Node* root){
        
        // Map to store the bottom view nodes based on their vertical positions
        map<int, int> mpp;
        queue<pair<Node*, int>> q;
        
        q.push({root, 0});
        while(!q.empty()){
            // Retrieve the node and its vertical position from the front of the queue
            auto it = q.front();
            q.pop();
            Node* node = it.first;
            int line = it.second;
            
            // Update the map with the node's data for the current vertical position
            mpp[line] = node->data;
            
            // Process left child
            if(node->left != NULL){
                // Push the left child with a decreased vertical position into the queue
                q.push({node->left, line - 1});
            }
            
            // Process right child
            if(node->right != NULL){
                // Push the right child with an increased vertical position into the queue
                q.push({node->right, line + 1});
            }
        }
        for(auto it : mpp){
            ans.push_back(it.second);
        }
        return ans;
    }
